export const RequestMaker = () => {
  const FONTS = [
    [1, "Font 1 (no name)", "inherit"],
    [2, "Font 2 (no name)", "inherit"],
    [3, "Sakura", "'Cherry Bomb One', cursive"],
    [4, "Jellybean", "'Chicle', cursive"],
    [5, "Font 5 (no name)", "inherit"],
    [6, "Modern", "'MuseoModerno', sans-serif"],
    [7, "Medieval", "'MedievalSharp', serif"],
    [8, "8Bit", "'Pixelify Sans', monospace"],
    [9, "Font 9 (no name)", "inherit"],
    [10, "Vampyre", "'Grenze Gotisch', serif"],
    [11, "gg sans", "'Noto Sans', sans-serif"],
    [12, "Tempo", "'Zilla Slab', serif"]
  ];
  const EFFECTS = [
    [1, "Solid", 1],
    [2, "Gradient", 5],
    [3, "Neon", 1],
    [4, "Toon", 1],
    [5, "Pop", 1],
    [6, "Glow", 5]
  ];
  const LANGS = [
    ["json", "JSON"],
    ["js", "JavaScript"],
    ["ts", "TypeScript"],
    ["py", "Python"],
    ["curl", "cURL"]
  ];

  useEffect(() => {
    if (document.getElementById("dns-preview-fonts")) return;
    const link = document.createElement("link");
    link.id = "dns-preview-fonts";
    link.rel = "stylesheet";
    link.href =
      "https://fonts.googleapis.com/css2?family=Cherry+Bomb+One&family=Chicle&family=MuseoModerno:wght@600&family=MedievalSharp&family=Pixelify+Sans:wght@600&family=Grenze+Gotisch:wght@600&family=Zilla+Slab:wght@600&family=Noto+Sans:wght@600&display=swap";
    document.head.appendChild(link);
  }, []);

  const [name, setName] = useState("MyBot");
  const [font, setFont] = useState(6);
  const [effect, setEffect] = useState(4);
  const [colors, setColors] = useState(["#F0C040", "#5865F2", "#F0C040", "#5865F2", "#F0C040"]);
  const [count, setCount] = useState(1);
  const [clear, setClear] = useState(false);
  const [lang, setLang] = useState("json");
  const [copied, setCopied] = useState(false);

  const maxColors = EFFECTS.find((e) => e[0] === effect)[2];
  const used = Math.min(count, maxColors);
  const shown = colors.slice(0, used);
  const toDec = (hex) => parseInt(hex.replace("#", ""), 16);
  const shade = (hex, f) => {
    const n = toDec(hex);
    const ch = (v) => Math.max(0, Math.min(255, Math.round(v * f))).toString(16).padStart(2, "0");
    return "#" + ch((n >> 16) & 255) + ch((n >> 8) & 255) + ch(n & 255);
  };

  const c0 = shown[0];
  const grad = shown.length > 1 ? "linear-gradient(90deg, " + shown.join(", ") + ")" : c0;
  const clipText = {
    backgroundImage: shown.length > 1 ? grad : "linear-gradient(" + c0 + ", " + c0 + ")",
    WebkitBackgroundClip: "text",
    backgroundClip: "text",
    color: "transparent"
  };
  const styleFor = (id) => ({
    1: { color: c0 },
    2: clipText,
    3: { color: "#FFFFFF", textShadow: "0 0 3px " + c0 + ", 0 0 8px " + c0 + ", 0 0 16px " + c0 },
    4: { color: c0, WebkitTextStroke: "1px #000000", textShadow: "2px 2px 0 #000000" },
    5: { color: c0, textShadow: "0 3px 0 " + shade(c0, 0.55) },
    6: Object.assign({}, clipText, { filter: "drop-shadow(0 0 6px " + c0 + ")" })
  }[id]);
  const effectStyle = styleFor(effect);

  const fontRow = FONTS.find((f) => f[0] === font);
  const nameStyle = clear
    ? { color: "#F2F3F5", fontFamily: "'Noto Sans', sans-serif" }
    : Object.assign({ fontFamily: fontRow[2] }, effectStyle);

  const body = clear
    ? { display_name_font_id: null, display_name_effect_id: null, display_name_colors: null }
    : {
        display_name_font_id: font,
        display_name_effect_id: effect,
        display_name_colors: shown.map(toDec)
      };

  const json = JSON.stringify(body, null, 2);
  const inline = JSON.stringify(body);
  const pyBody = json.replace(/null/g, "None").replace(/\n/g, "\n    ");
  const jsBody = json.replace(/\n/g, "\n  ");

  const code = {
    json: json,
    js:
      'const { Routes } = require("discord.js");\n\n' +
      "async function applyStyle(client, guildId) {\n" +
      "  const body = " + jsBody + ";\n" +
      "  const member = await client.rest.patch(Routes.guildMember(guildId), { body });\n" +
      "  console.log(member.display_name_styles);\n" +
      "}",
    ts:
      'import { Client, Routes } from "discord.js";\n\n' +
      "interface DisplayNameStyles {\n  font_id: number;\n  effect_id: number;\n  colors: number[];\n}\n\n" +
      "async function applyStyle(client: Client, guildId: string): Promise<DisplayNameStyles | null> {\n" +
      "  const body = " + jsBody + ";\n" +
      "  const member = (await client.rest.patch(Routes.guildMember(guildId), { body })) as {\n" +
      "    display_name_styles: DisplayNameStyles | null;\n  };\n" +
      "  return member.display_name_styles;\n" +
      "}",
    py:
      "import requests\n\n" +
      'TOKEN = "your-bot-token"\n' +
      'GUILD_ID = "your-guild-id"\n\n' +
      "def apply_style():\n" +
      "    body = " + pyBody + "\n" +
      "    r = requests.patch(\n" +
      '        f"https://discord.com/api/v10/guilds/{GUILD_ID}/members/@me",\n' +
      '        headers={"Authorization": f"Bot {TOKEN}"},\n' +
      "        json=body,\n    )\n" +
      "    r.raise_for_status()\n" +
      '    print(r.json().get("display_name_styles"))\n\n' +
      "apply_style()",
    curl:
      'curl -X PATCH "https://discord.com/api/v10/guilds/GUILD_ID/members/@me" \\\n' +
      '  -H "Authorization: Bot YOUR_TOKEN" \\\n' +
      '  -H "Content-Type: application/json" \\\n' +
      "  -d '" + inline + "'"
  }[lang];

  const copy = () => {
    navigator.clipboard.writeText(code).then(() => {
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    });
  };

  const setColorAt = (i, value) => {
    const next = colors.slice();
    next[i] = value;
    setColors(next);
  };

  const S = {
    panel: { background: "#2B2D31", border: "1px solid #3F4147", borderRadius: 8, padding: 14, marginBottom: 12 },
    label: { display: "block", fontSize: 12, fontWeight: 700, textTransform: "uppercase", letterSpacing: "0.02em", color: "#B5BAC1", marginBottom: 8 },
    input: { width: "100%", background: "#1E1F22", color: "#F2F3F5", border: "1px solid #1E1F22", borderRadius: 4, padding: "8px 10px", fontSize: 14, outline: "none" },
    btn: { background: "#4E5058", color: "#FFFFFF", border: "none", borderRadius: 4, padding: "6px 12px", fontSize: 13, fontWeight: 500, cursor: "pointer" },
    btnPrimary: { background: "#5865F2", color: "#FFFFFF", border: "none", borderRadius: 4, padding: "6px 12px", fontSize: 13, fontWeight: 500, cursor: "pointer" },
    swatch: { display: "flex", alignItems: "center", gap: 10, background: "#1E1F22", borderRadius: 6, padding: "6px 10px 6px 6px" },
    colorInput: { width: 36, height: 36, border: "none", borderRadius: 4, padding: 0, background: "transparent", cursor: "pointer" },
    mono: { fontFamily: "ui-monospace, SFMono-Regular, Menlo, monospace", fontSize: 12 },
    grid: { display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(96px, 1fr))", gap: 8 },
    tileName: { fontSize: 12, fontWeight: 600, color: "#DBDEE1", marginTop: 6 },
    tileId: { fontSize: 10, color: "#949BA4", marginTop: 2 }
  };
  const tile = (active) => ({
    background: active ? "#3C3F72" : "#1E1F22",
    border: active ? "2px solid #5865F2" : "2px solid #1E1F22",
    borderRadius: 8,
    padding: "12px 6px 10px",
    textAlign: "center",
    cursor: "pointer",
    overflow: "hidden"
  });

  return (
    <div className="not-prose">
      <div style={{ background: "#232428", border: "1px solid #3F4147", borderRadius: 8, padding: 16, marginBottom: 12 }}>
        <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
          <div style={{ width: 48, height: 48, borderRadius: "50%", background: "#5865F2", flexShrink: 0 }} />
          <div style={{ minWidth: 0 }}>
            <div style={Object.assign({ fontSize: 24, fontWeight: 600, lineHeight: 1.3, display: "inline-block", padding: "2px 4px" }, nameStyle)}>
              {name || "MyBot"}
            </div>
            <div style={{ fontSize: 12, color: "#B5BAC1" }}>
              {clear ? "No style" : fontRow[1] + " / " + EFFECTS.find((e) => e[0] === effect)[1]}
              <span style={{ marginLeft: 6, background: "#5865F2", color: "#fff", fontSize: 10, padding: "1px 4px", borderRadius: 3 }}>APP</span>
            </div>
          </div>
        </div>
      </div>

      <div style={S.panel}>
        <label style={S.label}>Preview Name</label>
        <input style={S.input} value={name} maxLength={32} onChange={(e) => setName(e.target.value)} />
        <label style={{ display: "flex", alignItems: "center", gap: 8, fontSize: 14, color: "#DBDEE1", marginTop: 10, cursor: "pointer" }}>
          <input type="checkbox" checked={clear} onChange={(e) => setClear(e.target.checked)} style={{ accentColor: "#5865F2", width: 16, height: 16 }} />
          Clear the style instead (send all three fields as null)
        </label>
      </div>

      {!clear && (
        <div>
          <div style={S.panel}>
            <label style={S.label}>Font</label>
            <div style={S.grid}>
              {FONTS.map((f) => (
                <button key={f[0]} style={tile(font === f[0])} onClick={() => setFont(f[0])} title={"Font " + f[0]}>
                  <div style={{ fontSize: 28, fontWeight: 600, lineHeight: 1.2, color: "#F2F3F5", fontFamily: f[2] }}>Aa</div>
                  <div style={S.tileName}>{f[1].replace(" (no name)", "")}</div>
                  <div style={S.tileId}>ID {f[0]}</div>
                </button>
              ))}
            </div>
          </div>

          <div style={S.panel}>
            <label style={S.label}>Effect</label>
            <div style={S.grid}>
              {EFFECTS.map((f) => (
                <button key={f[0]} style={tile(effect === f[0])} onClick={() => setEffect(f[0])} title={"Effect " + f[0]}>
                  <div style={Object.assign({ fontSize: 28, fontWeight: 700, lineHeight: 1.2, display: "inline-block", padding: "0 4px", fontFamily: fontRow[2] }, styleFor(f[0]))}>Aa</div>
                  <div style={S.tileName}>{f[1]}</div>
                  <div style={S.tileId}>ID {f[0]} · {f[2] === 1 ? "1 color" : "1 to 5 colors"}</div>
                </button>
              ))}
            </div>
          </div>

          <div style={S.panel}>
            <label style={S.label}>Colors ({used} of {maxColors})</label>
            <div style={{ display: "flex", flexWrap: "wrap", gap: 8 }}>
              {shown.map((c, i) => (
                <div key={i} style={S.swatch}>
                  <input type="color" value={c} style={S.colorInput} onChange={(e) => setColorAt(i, e.target.value.toUpperCase())} />
                  <div style={{ lineHeight: 1.3 }}>
                    <div style={Object.assign({ color: "#F2F3F5", fontWeight: 600 }, S.mono)}>{c}</div>
                    <div style={Object.assign({ color: "#949BA4" }, S.mono)}>{toDec(c)}</div>
                  </div>
                </div>
              ))}
            </div>
            {maxColors > 1 && (
              <div style={{ display: "flex", gap: 8, marginTop: 10 }}>
                <button style={Object.assign({}, S.btnPrimary, used >= 5 ? { opacity: 0.5, cursor: "not-allowed" } : {})} disabled={used >= 5} onClick={() => setCount(Math.min(5, used + 1))}>+ Add Color</button>
                <button style={Object.assign({}, S.btn, used <= 1 ? { opacity: 0.5, cursor: "not-allowed" } : {})} disabled={used <= 1} onClick={() => setCount(Math.max(1, used - 1))}>Remove Color</button>
              </div>
            )}
          </div>
        </div>
      )}

      <div style={{ background: "#111214", border: "1px solid #3F4147", borderRadius: 12, overflow: "hidden" }}>
        <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", background: "#1A1B1E", borderBottom: "1px solid #2B2D31", padding: "0 8px" }}>
          <div style={{ display: "flex", overflowX: "auto" }}>
            {LANGS.map((l) => {
              const active = lang === l[0];
              return (
                <button
                  key={l[0]}
                  onClick={() => setLang(l[0])}
                  style={{
                    background: "transparent",
                    border: "none",
                    borderBottom: active ? "2px solid #5865F2" : "2px solid transparent",
                    color: active ? "#FFFFFF" : "#949BA4",
                    fontSize: 13,
                    fontWeight: active ? 600 : 500,
                    padding: "10px 12px",
                    cursor: "pointer",
                    whiteSpace: "nowrap"
                  }}
                >
                  {l[1]}
                </button>
              );
            })}
          </div>
          <button
            onClick={copy}
            style={{ background: copied ? "#248046" : "#2B2D31", color: "#FFFFFF", border: "1px solid #3F4147", borderRadius: 6, fontSize: 12, padding: "4px 10px", cursor: "pointer", whiteSpace: "nowrap" }}
          >
            {copied ? "Copied!" : "Copy"}
          </button>
        </div>
        <pre style={Object.assign({ margin: 0, padding: 16, overflowX: "auto", color: "#DBDEE1", background: "#111214", whiteSpace: "pre", lineHeight: 1.6 }, S.mono, { fontSize: 13 })}>{code}</pre>
      </div>
    </div>
  );
};

The Request Maker builds a Display Name Styles request for you and shows a live preview of the name. Type your bot's name, pick a font, an effect, and colors, then copy the JSON body or the ready-made code in JavaScript, TypeScript, Python, or cURL. Only fonts and effects that save for bots are listed (see [Fonts](/fonts) and [Effects](/effects)).

<Note>
  The preview is an approximation, not Discord's exact rendering. Fonts use the same typefaces Discord uses where they are publicly available. Medieval and Vampyre use look-alike fonts, and unnamed fonts (1, 2, 5, 9) show in the default font.
</Note>

## Sending the Request

Send the body with `PATCH /guilds/{guild_id}/members/@me` using your bot token. Check `display_name_styles` in the response, because HTTP 200 alone does not confirm the style was saved. See [Endpoint](/endpoint) for the full reference and [Gotchas](/gotchas) for rate limits.
